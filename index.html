// Hilfsfunktion: Verkleinert das Bild für die KI (spart RAM & beschleunigt die Erkennung um das 5-fache) function scaleCanvasForAI(sourceCanvas, maxSide = 640) { const scale = Math.min(1, maxSide / Math.max(sourceCanvas.width, sourceCanvas.height)); const c = document.createElement("canvas"); c.width = Math.round(sourceCanvas.width * scale); c.height = Math.round(sourceCanvas.height * scale); const ctx = c.getContext("2d"); ctx.drawImage(sourceCanvas, 0, 0, c.width, c.height); return c; } // Optimierte Florence-2 Erkennungsfunktion async function processSlot(key) { const s = slots[key]; if (!s.sourceCanvas) { setStatus(key, "Bitte zuerst ein Foto aufnehmen.", "warn"); return; } if (s.busy) return; s.busy = true; setBadge($("slotBadge" + key.toUpperCase()), "Florence-2 analysiert …", "warn"); setStatus(key, "Florence-2 liest gebogenes Etikett …"); setProgress(key, 20); // Kurze Pause, damit der Browser die Statusmeldung zeichnen kann await new Promise(r => setTimeout(r, 100)); try { const { model, processor } = await getFlorence2(); setProgress(key, 40); await new Promise(r => setTimeout(r, 100)); // Bild für die KI verkleinern const aiCanvas = scaleCanvasForAI(s.sourceCanvas, 640); const image = await window.RawImage.fromCanvas(aiCanvas); const prompt = ''; const inputs = await processor(image, prompt); setProgress(key, 65); setStatus(key, "Berechne Text mit KI (bitte kurz warten) …"); await new Promise(r => setTimeout(r, 100)); // Schnelle Generierung mit max_new_tokens: 64 & num_beams: 1 const generated_ids = await model.generate({ ...inputs, max_new_tokens: 64, num_beams: 1 }); const rawText = processor.batch_decode(generated_ids, { skip_special_tokens: false })[0]; s.rawText = rawText; setProgress(key, 90); const { fields, values } = parseLabelFields(rawText); s.fields = fields; s.values = values; renderSlot(key); setProgress(key, 100); const hasData = Object.values(values).some(v => v !== ""); setStatus(key, hasData ? "Etikett per Florence-2 gelesen." : "Text gelesen, bitte Werte prüfen.", hasData ? "ok" : "warn"); setBadge($("slotBadge" + key.toUpperCase()), hasData ? "erkannt" : "unsicher", hasData ? "ok" : "warn"); compareSlots(); } catch (e) { console.error("Florence-2 Fehler:", e); setStatus(key, "Fehler bei der Erkennung: " + e.message, "bad"); setBadge($("slotBadge" + key.toUpperCase()), "Fehler", "bad"); setProgress(key, 0); } finally { s.busy = false; setTimeout(() => setProgress(key, 0), 1000); } }
